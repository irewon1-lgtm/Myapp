name: Chatbook cold-route correction with complete regression gate
on:
  push:
    branches: [chatbook-app-20260923]
    paths: [.github/workflows/chatbook-bootstrap-routing-correction.yml]
permissions:
  contents: write
concurrency:
  group: chatbook-live-publication
  cancel-in-progress: false
jobs:
  correct:
    runs-on: ubuntu-latest
    timeout-minutes: 35
    env:
      CHATBOOK_ROUTING_CORRECTION: '1'
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.sha }}
          fetch-depth: 0
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Apply only tested cold-route correction
        run: |
          test ! -f .deployment/live-routing-correction-receipt.json
          python scripts/bootstrap-routing-fix.py
          node scripts/live-contract.mjs check
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add netlify tests/live scripts/bootstrap-routing-once.mjs live/shell-contract.json
          git commit -m 'Handle bootstrap assets explicitly on cold Netlify routes'
      - run: npm ci --no-audit --no-fund
      - run: npm run build
      - run: node scripts/live-package.mjs
      - name: Prepare Chromium
        run: |
          python -m pip install playwright==1.55.0
          python -m playwright install --with-deps chromium
      - name: Verify without any static-file shortcut
        run: python tests/live/browser.py
      - name: Publish tested source and live assets
        env:
          CHATBOOK_BASE_SHA: ${{ github.sha }}
        run: node scripts/promote-live.mjs
      - name: Await separate single correction handoff
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: node scripts/bootstrap-routing-once.mjs
      - name: Verify actual HTTPS cold boot and offline restart
        run: python tests/live/production-final.py
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: chatbook-final-bootstrap-evidence
          retention-days: 3
          path: |
            review/live-browser/
            review/live-production/
            live/channel.json
            live/release.json
            live/shell-contract.json
